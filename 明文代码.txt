// 校验UUID格式
function 校验UUID格式(str) {
    const uuid正则 = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
    return uuid正则.test(str);
}

// 校验IP地址
function 校验IP地址(ip) {
    const ipv4正则 = /^(?:(?:25[0-5]|2[0-4][0-9]|[01]?[0-9][0-9]?)\.){3}(?:25[0-5]|2[0-4][0-9]|[01]?[0-9][0-9]?)$/;
    if (ipv4正则.test(ip)) return true;
    const ipv6正则 = /^(?:[0-9a-fA-F]{1,4}:){7}[0-9a-fA-F]{1,4}$/;
    if (ipv6正则.test(ip)) return true;
    const ipv6短正则 = /^::1$|^::$|^(?:[0-9a-fA-F]{1,4}:)*::(?:[0-9a-fA-F]{1,4}:)*[0-9a-fA-F]{1,4}$/;
    return ipv6短正则.test(ip);
}

// 创建节点命名器
function 创建节点命名器(跳过编号 = false) {
    const 强制跳过 = (typeof 优选IP来源URL !== 'undefined' && 优选IP来源URL && 优选IP来源URL.trim());
    let 跳过编号开关 = 强制跳过 || 跳过编号;
    const 计数器 = {};

    function 设置跳过编号(flag) { if (!强制跳过) 跳过编号开关 = flag; }
    function 命名器(基础名, 节点名 = null) {
        if (跳过编号开关 || (基础名 && 基础名.includes('.'))) return 节点名 || 基础名;
        if (!计数器[基础名]) 计数器[基础名] = 0;
        计数器[基础名]++;
        const 序号 = String(计数器[基础名]).padStart(2, '0');
        return `${节点名 || 基础名}-${序号}`;
    }
    return { 命名器, 设置跳过编号 };
}

// 加载KV配置
async function 加载KV配置(强制刷新 = false) {
    if (!KV存储) return;
    if (!强制刷新 && KV配置最后加载时间 > 0 && (Date.now() - KV配置最后加载时间) < KV缓存TTL) return;
    try {
        const 配置数据 = await KV存储.get('c');
        if (配置数据) KV配置 = JSON.parse(配置数据);
        KV配置最后加载时间 = Date.now();
    } catch (error) { KV配置 = {}; }
}

// 保存KV配置
async function 保存KV配置() {
    if (!KV存储) return;
    try { await KV存储.put('c', JSON.stringify(KV配置)); KV配置最后加载时间 = Date.now(); }
    catch (error) { throw error; }
}

// 智能地区选择
function 智能地区选择(worker地区, 可用IP列表, 使用地区匹配 = 启用地区匹配) {
    if (!使用地区匹配 || !worker地区) return 可用IP列表;
    const 优先级地区 = 按优先级获取地区(worker地区);
    const 排序后 = [];
    for (const 地区 of 优先级地区) {
        const 地区IP = 可用IP列表.filter(ip => ip.regionCode === 地区);
        排序后.push(...地区IP);
    }
    return 排序后;
}

// 解析地址和端口
function 解析地址和端口(输入) {
    if (输入.includes('[') && 输入.includes(']')) {
        const 匹配 = 输入.match(/^\[([^\]]+)\](?::(\d+))?$/);
        if (匹配) return { 地址: 匹配[1], 端口: 匹配[2] ? parseInt(匹配[2], 10) : null };
    }
    const 最后一个冒号索引 = 输入.lastIndexOf(':');
    if (最后一个冒号索引 > 0) {
        const 地址 = 输入.substring(0, 最后一个冒号索引);
        const 端口字符串 = 输入.substring(最后一个冒号索引 + 1);
        const 端口 = parseInt(端口字符串, 10);
        if (!地址.includes(':') && !isNaN(端口) && 端口 > 0 && 端口 <= 65535) return { 地址, 端口 };
    }
    return { 地址: 输入, 端口: null };
}

    if (启用ECH) { 响应头['X-ECH-Status'] = 'ENABLED'; if (ech配置) 响应头['X-ECH-Config-Length'] = String(ech配置.length); }
    return new Response(订阅内容, { headers: 响应头 });
}

function 解析链接为Clash节点(链接) {
    try {
        if (链接.startsWith('vless://')) {
            const url = new URL(链接);
            const 名称 = decodeURIComponent(url.hash.substring(1));
            const uuid = url.username;
            const 服务器 = url.hostname;
            const 端口 = parseInt(url.port) || 443;
            const 参数 = new URLSearchParams(url.search);
            const tls = 参数.get('security') === 'tls' || 参数.get('tls') === 'true';
            const 网络 = 参数.get('type') || 'ws';
            const 路径 = 参数.get('path') || '/?ed=2048';
            const 主机 = 参数.get('host') || 服务器;
            const sni = 参数.get('sni') || 主机;
            const alpn = 参数.get('alpn') || 'h3';
            const fingerprint = 参数.get('fp') || 参数.get('client-fingerprint') || 'chrome';
            const ech = 参数.get('ech');
            const 节点 = { name: 名称, type: 'vless', server: 服务器, port: 端口, uuid: uuid, tls: tls, network: 网络, 'client-fingerprint': fingerprint };
            if (tls) { 节点.servername = sni; 节点.alpn = alpn.split(',').map(a => a.trim()); 节点['skip-cert-verify'] = false; }
            if (网络 === 'ws') 节点['ws-opts'] = { path: 路径, headers: { Host: 主机 } };
            if (ech) 节点['ech-opts'] = { enable: true, 'query-server-name': 自定义ECH域名 || 'cloudflare-ech.com' };
            return 节点;
        }
        if (链接.startsWith('trojan://')) {
            const url = new URL(链接);
            const 名称 = decodeURIComponent(url.hash.substring(1));
            const 密码 = url.username;
            const 服务器 = url.hostname;
            const 端口 = parseInt(url.port) || 443;
            const 参数 = new URLSearchParams(url.search);
            const 网络 = 参数.get('type') || 'ws';
            const 路径 = 参数.get('path') || '/?ed=2048';
            const 主机 = 参数.get('host') || 服务器;
            const sni = 参数.get('sni') || 主机;
            const alpn = 参数.get('alpn') || 'h3';
            const ech = 参数.get('ech');
            const 节点 = { name: 名称, type: 'trojan', server: 服务器, port: 端口, password: 密码, network: 网络, sni: sni, alpn: alpn.split(',').map(a => a.trim()), 'skip-cert-verify': false };
            if (网络 === 'ws') 节点['ws-opts'] = { path: 路径, headers: { Host: 主机 } };
            if (ech) 节点['ech-opts'] = { enable: true, 'query-server-name': 自定义ECH域名 || 'cloudflare-ech.com' };
            return 节点;
        }
    } catch (e) { return null; }
    return null;
}

async function 生成Clash配置(链接列表, request, 用户) {
    const 订阅URL = new URL(request.url);
    订阅URL.pathname = 订阅URL.pathname.replace(/\/sub$/, '') + '/sub';
    订阅URL.searchParams.set('target', 'base64');
    const 编码URL = encodeURIComponent(订阅URL.toString());
    const 转换器URL = `${订阅转换地址}?target=clash&url=${编码URL}&insert=false&emoji=true&list=false&xudp=false&udp=false&tfo=false&expand=true&scv=false&fdn=false&new_name=true`;
    try {
        const 响应 = await fetch(转换器URL);
        if (!响应.ok) throw new Error('订阅转换服务失败');
        let clash配置 = await 响应.text();
        if (启用ECH) {
            clash配置 = clash配置.split('\n').map(行 => {
                if (/^\s*-\s*\{/.test(行) && 行.includes('name:') && 行.includes('server:')) {
                    if (行.includes('ech-opts')) return 行;
                    const 最后一个大括号 = 行.lastIndexOf('}');
                    if (最后一个大括号 > 0) {
                        const 之前 = 行.substring(0, 最后一个大括号).trim();
                        if (之前.length > 0) {
                            const ech域名 = 自定义ECH域名 || 'cloudflare-ech.com';
                            const 需要逗号 = !之前.endsWith(',') && !之前.endsWith('{');
                            return 行.substring(0, 最后一个大括号) + (需要逗号 ? ', ' : ' ') + `ech-opts: {enable: true, query-server-name: ${ech域名}}` + 行.substring(最后一个大括号);
                        }
                    }
                }
                return 行;
            }).join('\n');
        }
        clash配置 = clash配置.replace(/^(\s*nameserver:\s*\n)((?:\s*-\s*[^\n]+\n)*)/m, (匹配, 头, 项) => { const dns = 自定义DNS || 'https://223.5.5.5/dns-query'; return 头 + `    - ${dns}\n`; });
        return clash配置;
    } catch (e) { throw new Error('无法获取 Clash 配置: ' + e.message); }
}

let ECH调试信息 = '';
async function 获取ECH配置(域名) {
    if (!启用ECH) { ECH调试信息 = 'ECH功能已禁用'; return null; }
    ECH调试信息 = '';
    const 调试步骤 = [];
    try {
        调试步骤.push('尝试使用 Google DNS 查询 cloudflare-ech.com...');
        const ech域名URL = `https://v.recipes/dns/dns.google/dns-query?name=cloudflare-ech.com&type=65`;
        const ech响应 = await fetch(ech域名URL, { headers: { 'Accept': 'application/json' } });
        调试步骤.push(`Google DNS 响应状态: ${ech响应.status}`);
        if (ech响应.ok) {
            const ech数据 = await ech响应.json();
            调试步骤.push(`Google DNS 返回数据: ${JSON.stringify(ech数据).substring(0, 200)}...`);
            if (ech数据.Answer && ech数据.Answer.length > 0) {
                for (const 答案 of ech数据.Answer) {
                    if (答案.data) {
                        const 数据字符串 = typeof 答案.data === 'string' ? 答案.data : JSON.stringify(答案.data);
                        const ech匹配 = 数据字符串.match(/ech=([^\s"']+)/);
                        if (ech匹配 && ech匹配[1]) { ECH调试信息 = 调试步骤.join('\\n') + '\\n✅ 成功获取 ECH 配置'; return ech匹配[1]; }
                        try {
                            const 解码 = atob(答案.data);
                            const 解码匹配 = 解码.match(/ech=([^\s"']+)/);
                            if (解码匹配 && 解码匹配[1]) { ECH调试信息 = 调试步骤.join('\\n') + '\\n✅ 成功从 base64 解码获取 ECH 配置'; return 解码匹配[1]; }
                        } catch (e) { 调试步骤.push(`base64 解码失败: ${e.message}`); }
                    }
                }
            }
        }
        调试步骤.push('尝试使用 Google DNS 查询目标域名...');
        const dohURL = `https://v.recipes/dns/dns.google/dns-query?name=${encodeURIComponent(域名)}&type=65`;
        const doh响应 = await fetch(dohURL, { headers: { 'Accept': 'application/json' } });
        if (doh响应.ok) {
            const doh数据 = await doh响应.json();
            if (doh数据.Answer && doh数据.Answer.length > 0) {
                for (const 答案 of doh数据.Answer) {
                    if (答案.data) {
                        const 数据字符串 = typeof 答案.data === 'string' ? 答案.data : JSON.stringify(答案.data);
                        const ech匹配 = 数据字符串.match(/ech=([^\s"']+)/);
                        if (ech匹配 && ech匹配[1]) return ech匹配[1];
                    }
                }
            }
        }
        ECH调试信息 = 调试步骤.join('\\n') + '\\n❌ 未获取到ECH配置';
        return null;
    } catch (error) { ECH调试信息 = 调试步骤.join('\\n') + '\\n❌ 错误: ' + error.message; return null; }
}

function 从源生成VLESS链接(源列表, 用户, worker域名, ech配置 = null, 跳过编号 = false) {
    const CF_HTTP端口 = [80, 8080, 8880, 2052, 2082, 2086, 2095];
    const CF_HTTPS端口 = [443, 2053, 2083, 2087, 2096, 8443];
    const 默认HTTPS端口 = [443];
    const 默认HTTP端口 = 禁用非TLS ? [] : [80];
    const 链接列表 = [];
    const ws路径 = '/?ed=2048';
    const 协议 = atob('dmxlc3M=');
    const { 命名器, 设置跳过编号 } = 创建节点命名器(跳过编号);

    for (const 项 of 源列表) {
        let 节点基础名 = 项.isp.replace(/\s/g, '_');
        if (项.colo && 项.colo.trim()) 节点基础名 = `${节点基础名}-${项.colo.trim()}`;
        const 安全IP = 项.ip.includes(':') ? `[${项.ip}]` : 项.ip;
        let 端口列表 = [];
        if (项.port) {
            const 端口 = 项.port;
            if (CF_HTTPS端口.includes(端口)) 端口列表.push({ 端口, tls: true });
            else if (CF_HTTP端口.includes(端口)) { if (!禁用非TLS) 端口列表.push({ 端口, tls: false }); }
            else 端口列表.push({ 端口, tls: true });
        } else {
            默认HTTPS端口.forEach(p => 端口列表.push({ 端口: p, tls: true }));
            默认HTTP端口.forEach(p => 端口列表.push({ 端口: p, tls: false }));
        }
        for (const { 端口, tls } of 端口列表) {
            const 后缀 = tls ? '-WS-TLS' : '-WS';
            const 节点名 = `${节点基础名}-${端口}${后缀}`;
            let ws节点名 = 跳过编号 ? 节点名 : 命名器(节点基础名, 节点名);
            if (tls) {
                const ws参数 = new URLSearchParams({ encryption: 'none', security: 'tls', sni: worker域名, fp: 启用ECH ? 'chrome' : 'randomized', type: 'ws', host: worker域名, path: ws路径 });
                if (启用ECH) { ws参数.set('alpn', 'h3'); ws参数.set('ech', `${自定义ECH域名}+${自定义DNS}`); }
                链接列表.push(`${协议}://${用户}@${安全IP}:${端口}?${ws参数.toString()}#${encodeURIComponent(ws节点名)}`);
            } else {
                const ws参数 = new URLSearchParams({ encryption: 'none', security: 'none', type: 'ws', host: worker域名, path: ws路径 });
                链接列表.push(`${协议}://${用户}@${安全IP}:${端口}?${ws参数.toString()}#${encodeURIComponent(ws节点名)}`);
            }
        }
    }
    return 链接列表;
}

        let 匹配;
        while ((匹配 = 行正则.exec(html)) !== null) {

async function 请求优选API(urls, 默认端口 = '443', 超时 = 3000) {
    if (!urls?.length) return [];
    const 结果集 = new Set();
    await Promise.allSettled(urls.map(async (url) => {
        try {
            const 控制器 = new AbortController();
            const 定时器 = setTimeout(() => 控制器.abort(), 超时);
            const 响应 = await fetch(url, { signal: 控制器.signal });
            clearTimeout(定时器);
            let 文本 = '';
            try {
                const 缓冲区 = await 响应.arrayBuffer();
                const 内容类型 = (响应.headers.get('content-type') || '').toLowerCase();
                const 字符集 = 内容类型.match(/charset=([^\s;]+)/i)?.[1]?.toLowerCase() || '';
                let 解码器列表 = ['utf-8', 'gb2312'];
                if (字符集.includes('gb') || 字符集.includes('gbk') || 字符集.includes('gb2312')) 解码器列表 = ['gb2312', 'utf-8'];
                let 解码成功 = false;
                for (const 解码器 of 解码器列表) {
                    try {
                        const 解码 = new TextDecoder(解码器).decode(缓冲区);
                        if (解码 && 解码.length > 0 && !解码.includes('\ufffd')) { 文本 = 解码; 解码成功 = true; break; }
                    } catch (e) { continue; }
                }
                if (!解码成功) 文本 = await 响应.text();
                if (!文本 || 文本.trim().length === 0) return;
            } catch (e) { return; }
            const 行列表 = 文本.trim().split('\n').map(l => l.trim()).filter(l => l);
            const isCSV = 行列表.length > 1 && 行列表[0].includes(',');
            const IPV6模式 = /^[^\[\]]*:[^\[\]]*:[^\[\]]/;
            const 解析URL = new URL(url);
            if (!isCSV) {
                行列表.forEach(行 => {
                    const hash索引 = 行.indexOf('#');
                    const [主机部分, 备注部分] = hash索引 > -1 ? [行.substring(0, hash索引), 行.substring(hash索引)] : [行, ''];
                    let 有端口 = false;
                    if (主机部分.startsWith('[')) { 有端口 = /\]:(\d+)$/.test(主机部分); }
                    else { const colon索引 = 主机部分.lastIndexOf(':'); 有端口 = colon索引 > -1 && /^\d+$/.test(主机部分.substring(colon索引 + 1)); }
                    const 端口 = 解析URL.searchParams.get('port') || 默认端口;
                    结果集.add(有端口 ? 行 : `${主机部分}:${端口}${备注部分}`);
                });
            }
        } catch (e) {}
    }));
    return Array.from(结果集);
}

async function 处理配置API(request) {
    if (request.method === 'GET') {
        if (!KV存储) return new Response(JSON.stringify({ error: 'KV存储未配置', kvEnabled: false }), { status: 503, headers: { 'Content-Type': 'application/json' } });
        return new Response(JSON.stringify({ ...KV配置, kvEnabled: true }), { headers: { 'Content-Type': 'application/json' } });
    } else if (request.method === 'POST') {
        if (!KV存储) return new Response(JSON.stringify({ success: false, message: 'KV存储未配置' }), { status: 503, headers: { 'Content-Type': 'application/json' } });
        try {
            const 新配置 = await request.json();
            for (const [键, 值] of Object.entries(新配置)) {
                if (值 === '' || 值 === null || 值 === undefined) delete KV配置[键];
                else KV配置[键] = 值;
            }
            await 保存KV配置();
            更新配置变量();
            if (新配置.yx !== undefined) 从YX更新优选列表();
            return new Response(JSON.stringify({ success: true, message: '配置已保存', config: KV配置 }), { headers: { 'Content-Type': 'application/json' } });

            return new Response(JSON.stringify({ success: false, message: '保存失败: ' + error.message }), { status: 500, headers: { 'Content-Type': 'application/json' } });
        }
    }
    return new Response(JSON.stringify({ error: 'Method not allowed' }), { status: 405, headers: { 'Content-Type': 'application/json' } });
}

function 解析YX为数组(yx值) {

function 解析SOCKS5配置(地址) {
    let [后部, 前部] = 地址.split("@").reverse();
    let 用户名, 密码, 主机名, socks端口;
    if (前部) { const 分割 = 前部.split(":"); if (分割.length !== 2) throw new Error(错误_无效SOCKS地址); [用户名, 密码] = 分割; }
    const 后部分割 = 后部.split(":");
    socks端口 = Number(后部分割.pop());
    if (isNaN(socks端口)) throw new Error(错误_无效SOCKS地址);

    if (主机名.includes(":") && !/^\[.*\]$/.test(主机名)) throw new Error(错误_无效SOCKS地址);

}

async function 处理WebSocket请求(request) {

    let 请求Socks配置 = null;
    if (请求Socks字符串) {
        try { 请求Socks配置 = 解析SOCKS5配置(请求Socks字符串); } catch (_) {}
    }

    const WS套接字对 = new WebSocketPair();
    const [客户端套接字, 服务端套接字] = Object.values(WS套接字对);
    服务端套接字.accept();

    let 远端连接包装 = { socket: null };
    let 是否DNS查询 = false;
    let 协议类型 = null;

    const 早期数据 = request.headers.get('sec-websocket-protocol') || '';
    const 可读流 = 创建WebSocket可读流(服务端套接字, 早期数据);

    可读流.pipeTo(new WritableStream({
        async write(块) {
            if (是否DNS查询) return await 转发UDP数据(块, 服务端套接字, null);
            if (远端连接包装.socket) {
                const 写入器 = 远端连接包装.socket.writable.getWriter();
                await 写入器.write(块);
                写入器.releaseLock();
                return;
            }
            if (!协议类型) {
                if (启用VLESS && 块.byteLength >= 24) {
                    const vless结果 = 解析WS首包(块, 用户UUID);
                    if (!vless结果.hasError) {
                        协议类型 = 'vless';
                        const { addressType, port, hostname, rawIndex, version, isUDP } = vless结果;
                        if (isUDP) {
                            if (port === 53) 是否DNS查询 = true;
                            else throw new Error(错误_UDP仅DNS);
                        }
                        const 响应头 = new Uint8Array([version[0], 0]);
                        const 原始数据 = 块.slice(rawIndex);
                        if (是否DNS查询) return 转发UDP数据(原始数据, 服务端套接字, 响应头);
                        await 转发TCP连接(addressType, hostname, port, 原始数据, 服务端套接字, 响应头, 远端连接包装, 请求备用, 生效地区, 请求地区匹配, 请求Socks配置);

                    }
                }
                if (启用Trojan && 块.byteLength >= 56) {
                    const trojan结果 = await 解析Trojan首包(块, 用户UUID);
                    if (!trojan结果.hasError) {

                        return;
                    }
                }
                throw new Error('Invalid protocol or authentication failed');
            }
        },
    })).catch((err) => {});

    return new Response(null, { status: 101, webSocket: 客户端套接字 });
}

function 创建WebSocket可读流(socket, earlyDataHeader) {
    let 已取消 = false;
    return new ReadableStream({
        start(控制器) {
            socket.addEventListener('message', (事件) => { if (!已取消) 控制器.enqueue(事件.data); });
            socket.addEventListener('close', () => { if (!已取消) { 关闭Socket静默(socket); 控制器.close(); } });
            socket.addEventListener('error', (err) => 控制器.error(err));
            const { earlyData, error } = base64转数组(earlyDataHeader);
            if (error) 控制器.error(error);
            else if (earlyData) 控制器.enqueue(earlyData);
        },
        cancel() { 已取消 = true; 关闭Socket静默(socket); }
    });
}

function base64转数组(b64Str) {
    if (!b64Str) return { error: null };
    try {
        b64Str = b64Str.replace(/-/g, '+').replace(/_/g, '/');
        return { earlyData: Uint8Array.from(atob(b64Str), (c) => c.charCodeAt(0)).buffer, error: null };
    } catch (error) { return { error }; }
}

function 解析WS首包(块, token) {
    if (块.byteLength < 24) return { hasError: true, message: 错误_无效数据 };
    const version = new Uint8Array(块.slice(0, 1));
    if (格式化UUID(new Uint8Array(块.slice(1, 17))) !== token) return { hasError: true, message: 错误_无效用户 };
    const optLen = new Uint8Array(块.slice(17, 18))[0];
    const cmd = new Uint8Array(块.slice(18 + optLen, 19 + optLen))[0];
    let isUDP = false;
    if (cmd === 1) {} else if (cmd === 2) { isUDP = true; } else { return { hasError: true, message: 错误_不支持命令 }; }
    const portIdx = 19 + optLen;
    const port = new DataView(块.slice(portIdx, portIdx + 2)).getUint16(0);
    let addrIdx = portIdx + 2, addrLen = 0, addrValIdx = addrIdx + 1, hostname = '';
    const addressType = new Uint8Array(块.slice(addrIdx, addrValIdx))[0];
    switch (addressType) {
        case 地址类型_IPV4: addrLen = 4; hostname = new Uint8Array(块.slice(addrValIdx, addrValIdx + addrLen)).join('.'); break;
        case 地址类型_URL: addrLen = new Uint8Array(块.slice(addrValIdx, addrValIdx + 1))[0]; addrValIdx += 1; hostname = new TextDecoder().decode(块.slice(addrValIdx, addrValIdx + addrLen)); break;
        case 地址类型_IPV6: addrLen = 16; const ipv6 = []; const ipv6View = new DataView(块.slice(addrValIdx, addrValIdx + addrLen)); for (let i = 0; i < 8; i++) ipv6.push(ipv6View.getUint16(i * 2).toString(16)); hostname = ipv6.join(':'); break;
        default: return { hasError: true, message: `${错误_无效地址类型}: ${addressType}` };
    }
    if (!hostname) return { hasError: true, message: `${错误_地址为空}: ${addressType}` };
    return { hasError: false, addressType, port, hostname, isUDP, rawIndex: addrValIdx + addrLen, version };
}

async function 连接流(远端套接字, webSocket, 头数据, 重试函数) {
    let 头 = 头数据, 有数据 = false;
    await 远端套接字.readable.pipeTo(
        new WritableStream({
            async write(块, 控制器) {
                有数据 = true;
                if (webSocket.readyState !== 1) 控制器.error(错误_WS未打开);
                if (头) { webSocket.send(await new Blob([头, 块]).arrayBuffer()); 头 = null; }
                else { webSocket.send(块); }
            },
            abort(reason) {},
        })
    ).catch((error) => { 关闭Socket静默(webSocket); });
    if (!有数据 && 重试函数) 重试函数();
}

async function 转发UDP数据(udp块, webSocket, 响应头) {

        }));

}

async function 建立SOCKS5连接(addrType, address, port, socksConfig = 解析后的SOCKS5配置) {
    const { username, password, hostname, socksPort } = socksConfig;
    const socket = connect({ hostname, port: socksPort });
    const writer = socket.writable.getWriter();
    await writer.write(new Uint8Array(username ? [5, 2, 0, 2] : [5, 1, 0]));
    const reader = socket.readable.getReader();
    let res = (await reader.read()).value;
    if (res[0] !== 5 || res[1] === 255) throw new Error(错误_SOCKS无方法);
    if (res[1] === 2) {
        if (!username || !password) throw new Error(错误_SOCKS需要认证);
        const encoder = new TextEncoder();
        const authRequest = new Uint8Array([1, username.length, ...encoder.encode(username), password.length, ...encoder.encode(password)]);
        await writer.write(authRequest);
        res = (await reader.read()).value;
        if (res[0] !== 1 || res[1] !== 0) throw new Error(错误_SOCKS认证失败);
    }
    const encoder = new TextEncoder(); let DSTADDR;
    switch (addrType) {
        case 地址类型_IPV4: DSTADDR = new Uint8Array([1, ...address.split('.').map(Number)]); break;
        case 地址类型_URL: DSTADDR = new Uint8Array([3, address.length, ...encoder.encode(address)]); break;
        case 地址类型_IPV6: DSTADDR = new Uint8Array([4, ...address.split(':').flatMap(x => [parseInt(x.slice(0,2),16), parseInt(x.slice(2),16)])]); break;
        default: throw new Error(错误_无效地址类型);
    }
    await writer.write(new Uint8Array([5, 1, 0, ...DSTADDR, port >> 8, port & 255]));
    res = (await reader.read()).value;
    if (res[1] !== 0) throw new Error(错误_SOCKS连接失败);
    writer.releaseLock(); reader.releaseLock();
    return socket;
}

}

function 合并类型数组(first, ...args) {
    let len = first.length;
    for (let a of args) len += a.length;
    const r = new first.constructor(len);
    r.set(first, 0);
    len = first.length;
    for (let a of args) { r.set(a, len); len += a.length; }
    return r;
}

function 解析XHTTP_UUID(uuid) {
    uuid = uuid.replaceAll('-', '');
    const r = [];
    for (let index = 0; index < 16; index++) {
        const v = parseInt(uuid.substr(index * 2, 2), 16);
        r.push(v);
    }
    return r;
}

function 获取XHTTP缓冲区(size) { return new Uint8Array(new ArrayBuffer(size || XHTTP缓冲区大小)); }

async function 读取XHTTP首包(可读流, uuid字符串) {
    const 读取器 = 可读流.getReader({ mode: 'byob' });
    try {
        let r = await 读取器.readAtLeast(1 + 16 + 1, 获取XHTTP缓冲区());
        let rlen = 0;
        let idx = 0;
        let 缓存 = r.value;
        rlen += r.value.length;

        const version = 缓存[0];
        const id = 缓存.slice(1, 1 + 16);
        const uuid字节 = 解析XHTTP_UUID(uuid字符串);
        if (!校验XHTTP_UUID(id, uuid字节)) return `invalid UUID`;

        const pb_len = 缓存[1 + 16];
        const addr_plus1 = 1 + 16 + 1 + pb_len + 1 + 2 + 1;

        if (addr_plus1 + 1 > rlen) {
            if (r.done) return `header too short`;
            idx = addr_plus1 + 1 - rlen;
            r = await 读取器.readAtLeast(idx, 获取XHTTP缓冲区());
            rlen += r.value.length;
            缓存 = 合并类型数组(缓存, r.value);
        }

        const cmd = 缓存[1 + 16 + 1 + pb_len];
        if (cmd !== 1) return `unsupported command: ${cmd}`;

        const port = (缓存[addr_plus1 - 1 - 2] << 8) + 缓存[addr_plus1 - 1 - 1];
        const atype = 缓存[addr_plus1 - 1];
        let header_len = -1;
        if (atype === 地址类型_IPV4) header_len = addr_plus1 + 4;
        else if (atype === 地址类型_IPV6) header_len = addr_plus1 + 16;
        else if (atype === 地址类型_URL) header_len = addr_plus1 + 1 + 缓存[addr_plus1];

        if (header_len < 0) return 'read address type failed';

        idx = header_len - rlen;
        if (idx > 0) {
            if (r.done) return `read address failed`;
            r = await 读取器.readAtLeast(idx, 获取XHTTP缓冲区());
            rlen += r.value.length;
            缓存 = 合并类型数组(缓存, r.value);
        }

        let hostname = '';
        idx = addr_plus1;
        switch (atype) {
            case 地址类型_IPV4:
                hostname = 缓存.slice(idx, idx + 4).join('.');
                break;
            case 地址类型_URL:
                hostname = new TextDecoder().decode(缓存.slice(idx + 1, idx + 1 + 缓存[idx]));
                break;
            case 地址类型_IPV6:
                hostname = 缓存.slice(idx, idx + 16).reduce((s, b2, i2, a) => i2 % 2 ? s.concat(((a[i2-1] << 8) + b2).toString(16)) : s, []).join(':');
                break;
        }
        if (hostname.length < 1) return 'failed to parse hostname';

        const data = 缓存.slice(header_len);
        return { hostname, port, data, resp: new Uint8Array([version, 0]), reader: 读取器, done: r.done };
    } catch (error) {
        try { 读取器.releaseLock(); } catch (_) {}
        throw error;
    }
}

async function 上传到远端XHTTP(计数器, 写入器, httpx) {
    async function 内部上传(d) {
        if (!d || d.length === 0) return;
        计数器.add(d.length);
        try { await 写入器.write(d); } catch (error) { throw error; }
    }
    try {
        await 内部上传(httpx.data);
        let chunkCount = 0;
        while (!httpx.done) {
            const r = await httpx.reader.read(获取XHTTP缓冲区());
            if (r.done) break;
            await 内部上传(r.value);
            httpx.done = r.done;
            chunkCount++;
            if (chunkCount % 10 === 0) await XHTTP延迟(0);
            if (!r.value || r.value.length === 0) await XHTTP延迟(2);
        }
    } catch (error) { throw error; }
}

function 创建XHTTP上传器(httpx, 可写流) {
    const 计数器 = new XHTTP计数器();
    const 写入器 = 可写流.getWriter();
    const done = (async () => {
        try { await 上传到远端XHTTP(计数器, 写入器, httpx); }
        catch (error) { throw error; }
        finally { try { await 写入器.close(); } catch (error) {} }
    })();
    return { counter: 计数器, done, abort: () => { try { 写入器.abort(); } catch (_) {} } };
}

function 创建XHTTP下载器(resp, 远端可读流) {
    const 计数器 = new XHTTP计数器();
    let stream;
    const done = new Promise((resolve, reject) => {
        stream = new TransformStream({
            start(controller) {
                计数器.add(resp.length);
                controller.enqueue(resp);
            },
            transform(chunk, controller) {
                计数器.add(chunk.length);
                controller.enqueue(chunk);
            },
            cancel(reason) { reject(`download cancelled: ${reason}`); }
        }, null, new ByteLengthQueuingStrategy({ highWaterMark: XHTTP缓冲区大小 }));

        let lastActivity = Date.now();
        const idleTimer = setInterval(() => {
            if (Date.now() - lastActivity > 空闲超时毫秒) {
                try { stream.writable.abort?.('idle timeout'); } catch (_) {}
                clearInterval(idleTimer);
                reject('idle timeout');
            }
        }, 5000);

        const reader = 远端可读流.getReader();
        const writer = stream.writable.getWriter();
        (async () => {
            try {
                let chunkCount = 0;
                while (true) {
                    const r = await reader.read();
                    if (r.done) break;
                    lastActivity = Date.now();
                    await writer.write(r.value);
                    chunkCount++;
                    if (chunkCount % 5 === 0) await XHTTP延迟(0);
                }
                await writer.close();
                resolve();
            } catch (err) { reject(err); }
            finally {
                try { reader.releaseLock(); } catch (_) {}
                try { writer.releaseLock(); } catch (_) {}
                clearInterval(idleTimer);
            }
        })();
    });
    return { readable: stream.readable, counter: 计数器, done, abort: () => { try { stream.readable.cancel(); } catch (_) {} try { stream.writable.abort(); } catch (_) {} } };
}

async function 连接到远端XHTTP(httpx, ...备选列表) {
    let attempt = 0;
    let lastErr;
    const 连接列表 = [httpx.hostname, ...备选列表.filter(r => r && r !== httpx.hostname)];
    for (const hostname of 连接列表) {
        if (!hostname) continue;
        attempt = 0;
        while (attempt < 最大重试次数) {
            attempt++;
            try {
                const remote = connect({ hostname, port: httpx.port });
                const timeoutPromise = XHTTP延迟(连接超时毫秒).then(() => { throw new Error(atob('Y29ubmVjdCB0aW1lb3V0')); });
                await Promise.race([remote.opened, timeoutPromise]);
                const uploader = 创建XHTTP上传器(httpx, remote.writable);
                const downloader = 创建XHTTP下载器(httpx.resp, remote.readable);
                return { downloader, uploader, close: () => { try { remote.close(); } catch (_) {} } };
            } catch (err) {
                lastErr = err;
                if (attempt < 最大重试次数) await XHTTP延迟(500 * attempt);
            }
        }
    }
    return null;
}

async function 处理XHTTP客户端(body, uuid) {
    if (活跃连接数 >= 最大并发连接) return new Response('Too many connections', { status: 429 });
    活跃连接数++;
    let cleaned = false;
    const cleanup = () => { if (!cleaned) { 活跃连接数 = Math.max(0, 活跃连接数 - 1); cleaned = true; } };
    try {
        const httpx = await 读取XHTTP首包(body, uuid);
        if (typeof httpx !== 'object' || !httpx) return null;
        const remoteConnection = await 连接到远端XHTTP(httpx, 备用地址, '13.230.34.30');
        if (remoteConnection === null) return null;
        const connectionClosed = Promise.race([
            (async () => { try { await remoteConnection.downloader.done; } catch (err) {} })(),
            (async () => { try { await remoteConnection.uploader.done; } catch (err) {} })(),
            XHTTP延迟(空闲超时毫秒).then(() => {})
        ]).finally(() => {
            try { remoteConnection.close(); } catch (_) {}
            try { remoteConnection.downloader.abort(); } catch (_) {}
            try { remoteConnection.uploader.abort(); } catch (_) {}
            cleanup();
        });
        return { readable: remoteConnection.downloader.readable, closed: connectionClosed };
    } catch (error) { cleanup(); return null; }
}

    catch (err) { return null; }
}

// ==================== Trojan 首包解析 ====================
async function 解析Trojan首包(buffer, 明文密码) {
    const data = 数据转Uint8Array(buffer);
    const sha224密码 = await sha224哈希(明文密码);
    if (data.byteLength < 58) return { hasError: true, message: "invalid data" };
    let crLfIndex = 56;
    if (data[crLfIndex] !== 0x0d || data[crLfIndex + 1] !== 0x0a) return { hasError: true, message: "invalid header format" };
    for (let i = 0; i < crLfIndex; i++) {
        if (data[i] !== sha224密码.charCodeAt(i)) return { hasError: true, message: "invalid password" };
    }
    const socks5Index = crLfIndex + 2;
    if (data.byteLength < socks5Index + 6) return { hasError: true, message: "invalid S5 request data" };
    const cmd = data[socks5Index];
    if (cmd !== 1 && cmd !== 3) return { hasError: true, message: "unsupported command, only TCP/UDP is allowed" };
    const isUDP = cmd === 3;
    const atype = data[socks5Index + 1];
    let addressLength = 0;
    let addressIndex = socks5Index + 2;
    let address = "";
    switch (atype) {
        case 1: // IPv4
            addressLength = 4;
            if (data.byteLength < addressIndex + addressLength + 4) return { hasError: true, message: "invalid S5 request data" };
            address = `${data[addressIndex]}.${data[addressIndex + 1]}.${data[addressIndex + 2]}.${data[addressIndex + 3]}`;
            break;
        case 3: // Domain
            if (data.byteLength < addressIndex + 1) return { hasError: true, message: "invalid S5 request data" };
            addressLength = data[addressIndex];
            addressIndex += 1;
            if (data.byteLength < addressIndex + addressLength + 4) return { hasError: true, message: "invalid S5 request data" };
            address = new TextDecoder().decode(data.subarray(addressIndex, addressIndex + addressLength));
            break;
        case 4: // IPv6
            addressLength = 16;
            if (data.byteLength < addressIndex + addressLength + 4) return { hasError: true, message: "invalid S5 request data" };
            const ipv6 = [];
            for (let i = 0; i < 8; i++) {
                const partIndex = addressIndex + i * 2;
                ipv6.push(((data[partIndex] << 8) | data[partIndex + 1]).toString(16));
            }
            address = ipv6.join(":");
            break;
        default: return { hasError: true, message: `invalid addressType is ${atype}` };
    }
    if (!address) return { hasError: true, message: `address is empty, addressType is ${atype}` };
    const portIndex = addressIndex + addressLength;
    if (data.byteLength < portIndex + 4) return { hasError: true, message: "invalid S5 request data" };
    const portRemote = (data[portIndex] << 8) | data[portIndex + 1];
    return {
        hasError: false,
        addressType: atype,
        port: portRemote,
        hostname: address,
        isUDP,
        rawClientData: data.subarray(portIndex + 4)
    };
}

function 数据转Uint8Array(data) {
    if (data instanceof Uint8Array) return data;
    if (data instanceof ArrayBuffer) return new Uint8Array(data);
    if (ArrayBuffer.isView(data)) return new Uint8Array(data.buffer, data.byteOffset, data.byteLength);
    return new Uint8Array(data || 0);
}

// ==================== SHA224 哈希 ====================
async function sha224哈希(text) {

    const data = encoder.encode(text);

    ];
    const rightRotate = (value, amount) => (value >>> amount) | (value << (32 - amount));
    let H = [0xc1059ed8, 0x367cd507, 0x3070dd17, 0xf70e5939, 0xffc00b31, 0x68581511, 0x64f98fa7, 0xbefa4fa4];
    const msgLen = data.length;
    const bitLen = msgLen * 8;
    const paddedLen = Math.ceil((msgLen + 9) / 64) * 64;
    const padded = new Uint8Array(paddedLen);
    padded.set(data);
    padded[msgLen] = 0x80;
    const view = new DataView(padded.buffer);
    view.setUint32(paddedLen - 4, bitLen, false);
    for (let chunk = 0; chunk < paddedLen; chunk += 64) {
        const W = new Uint32Array(64);
        for (let i = 0; i < 16; i++) W[i] = view.getUint32(chunk + i * 4, false);
        for (let i = 16; i < 64; i++) {
            const s0 = rightRotate(W[i - 15], 7) ^ rightRotate(W[i - 15], 18) ^ (W[i - 15] >>> 3);
            const s1 = rightRotate(W[i - 2], 17) ^ rightRotate(W[i - 2], 19) ^ (W[i - 2] >>> 10);
            W[i] = (W[i - 16] + s0 + W[i - 7] + s1) >>> 0;
        }
        let [a, b, c, d, e, f, g, h] = H;
        for (let i = 0; i < 64; i++) {
            const S1 = rightRotate(e, 6) ^ rightRotate(e, 11) ^ rightRotate(e, 25);
            const ch = (e & f) ^ (~e & g);
            const temp1 = (h + S1 + ch + K[i] + W[i]) >>> 0;
            const S0 = rightRotate(a, 2) ^ rightRotate(a, 13) ^ rightRotate(a, 22);
            const maj = (a & b) ^ (a & c) ^ (b & c);
            const temp2 = (S0 + maj) >>> 0;
            h = g; g = f; f = e; e = (d + temp1) >>> 0; d = c; c = b; b = a; a = (temp1 + temp2) >>> 0;
        }
        H[0] = (H[0] + a) >>> 0; H[1] = (H[1] + b) >>> 0; H[2] = (H[2] + c) >>> 0;
        H[3] = (H[3] + d) >>> 0; H[4] = (H[4] + e) >>> 0; H[5] = (H[5] + f) >>> 0;
        H[6] = (H[6] + g) >>> 0; H[7] = (H[7] + h) >>> 0;
    }
    const result = [];
    for (let i = 0; i < 7; i++) {
        result.push(((H[i] >>> 24) & 0xff).toString(16).padStart(2, '0'));
        result.push(((H[i] >>> 16) & 0xff).toString(16).padStart(2, '0'));
        result.push(((H[i] >>> 8) & 0xff).toString(16).padStart(2, '0'));
        result.push((H[i] & 0xff).toString(16).padStart(2, '0'));
    }
    return result.join('');
}

            function showToast(message, type, options) {
                options = options || {};
                var stack = document.getElementById('cpToastStack');
                if (!stack) return;
                var typeMap = { success: '✓', info: '⌬', warn: '⚠', error: '✕' };
                var titleMap = { success: 'SUCCESS', info: 'INFO', warn: 'WARN', error: 'ERROR' };
                type = typeMap[type] ? type : 'success';
                var duration = options.duration || 3200;
                var toast = document.createElement('div');
                toast.className = 'cp-toast cp-toast-' + type;
                toast.style.setProperty('--cp-toast-dur', duration + 'ms');
                var icon = document.createElement('span');
                icon.className = 'cp-toast-icon';
                icon.textContent = typeMap[type];
                var body = document.createElement('div');
                body.className = 'cp-toast-body';
                var title = document.createElement('div');
                title.className = 'cp-toast-title';
                title.textContent = options.title || titleMap[type];
                var msg = document.createElement('div');
                msg.className = 'cp-toast-msg';
                msg.textContent = String(message == null ? '' : message);
                body.appendChild(title);
                body.appendChild(msg);
                var close = document.createElement('button');
                close.type = 'button';
                close.className = 'cp-toast-close';
                close.setAttribute('aria-label', 'close');
                close.textContent = '✕';
                toast.appendChild(icon);
                toast.appendChild(body);
                toast.appendChild(close);
                stack.appendChild(toast);
                requestAnimationFrame(function() { toast.classList.add('cp-show'); });
                var dismissed = false;
                function dismiss() {
                    if (dismissed) return;
                    dismissed = true;
                    toast.classList.remove('cp-show');
                    toast.classList.add('cp-hide');
                    setTimeout(function() {
                        if (toast.parentNode) toast.parentNode.removeChild(toast);
                    }, 400);
                }
                close.addEventListener('click', dismiss);
                var timer = setTimeout(dismiss, duration);
                toast.addEventListener('mouseenter', function() { clearTimeout(timer); });
                toast.addEventListener('mouseleave', function() { timer = setTimeout(dismiss, 1200); });
                return { dismiss: dismiss, element: toast };
            }
            window.cpToast = showToast;

            function tryOpenApp(schemeUrl, fallbackCallback, timeout) {
                timeout = timeout || 2500;
                var appOpened = false;
                var callbackExecuted = false;
                var startTime = Date.now();

                var blurHandler = function() {
                    var elapsed = Date.now() - startTime;
                    if (elapsed < 3000 && !callbackExecuted) {
                        appOpened = true;
                    }
                };

                window.addEventListener('blur', blurHandler);

                var hiddenHandler = function() {
                    var elapsed = Date.now() - startTime;
                    if (elapsed < 3000 && !callbackExecuted) {
                        appOpened = true;
                    }
                };

                document.addEventListener('visibilitychange', hiddenHandler);

                var iframe = document.createElement('iframe');
                iframe.style.display = 'none';
                iframe.style.width = '1px';
                iframe.style.height = '1px';
                iframe.src = schemeUrl;
                document.body.appendChild(iframe);

                setTimeout(function() {
                    iframe.parentNode && iframe.parentNode.removeChild(iframe);
                    window.removeEventListener('blur', blurHandler);
                    document.removeEventListener('visibilitychange', hiddenHandler);

                    if (!callbackExecuted) {
                        callbackExecuted = true;
                        if (!appOpened && fallbackCallback) {
                            fallbackCallback();
                        }
                    }
                }, timeout);
            }

                    for (let j = 0; j < charCount; j++) {
                        const char = cyberChars[Math.floor(Math.random() * cyberChars.length)];
                        const useAccent = Math.random() > 0.85;
                        const color = useAccent ? palette[Math.floor(Math.random() * palette.length)] : '';
                        text += color
                            ? ('<span style="color:' + color + ';text-shadow:0 0 8px ' + color + ';">' + char + '</span><br>')
                            : ('<span>' + char + '</span><br>');
                    }
                    column.innerHTML = text;
                    matrixContainer.appendChild(column);
                }

                setInterval(function() {
                    const cols = matrixContainer.querySelectorAll('.matrix-column');
                    cols.forEach(function(column) {
                        if (Math.random() > 0.94) {
                            const chars = column.querySelectorAll('span');
                            if (chars.length > 0) {
                                const target = chars[Math.floor(Math.random() * chars.length)];
                                const prev = target.style.color;
                                target.style.color = '#ffffff';
                                target.style.textShadow = '0 0 10px #ffffff, 0 0 18px #00f0ff';
                                setTimeout(function() {
                                    target.style.color = prev;
                                    target.style.textShadow = '';
                                }, 200);
                            }
                        }

async function checkECHStatus() {
    const echStatusEl = document.getElementById('echStatus');
    if (!echStatusEl) return;
    try {
        // 正确构建订阅 URL：移除当前路径末尾可能的斜杠，再拼接 /sub
        let basePath = window.location.pathname;
        if (basePath.endsWith('/')) basePath = basePath.slice(0, -1);
        const subscriptionUrl = window.location.origin + basePath + '/sub';
        // 添加随机参数避免缓存
        const urlWithCache = subscriptionUrl + (subscriptionUrl.includes('?') ? '&' : '?') + '_=' + Date.now();
        const response = await fetch(urlWithCache, { method: 'HEAD', cache: 'no-store' });
        const echStatusHeader = response.headers.get('X-ECH-Status');
        if (echStatusHeader === 'ENABLED') {
            echStatusEl.innerHTML = 'ECH状态: <span style="color: #00ff9d;">✅ 已启用</span>';
        } else {
            echStatusEl.innerHTML = 'ECH状态: <span style="color: #ffb400;">⚠️ 未启用</span>';
        }
    } catch (error) {
        echStatusEl.innerHTML = 'ECH状态: <span style="color: #ff3860;">❌ 检测失败</span>';
    }
}

            async function loadCurrentConfig() {
                const apiUrl = window.location.pathname + '/api/config';
                try {
                    const response = await fetch(apiUrl);
                    if (response.status === 503) {
                        document.getElementById('currentConfig').textContent = 'KV存储未配置，无法加载配置';
                        return;
                    }
                    if (!response.ok) throw new Error(await response.text());
                    const config = await response.json();
                    const displayConfig = {};
                    for (const [key, value] of Object.entries(config)) {
                        if (key !== 'kvEnabled') displayConfig[key] = value;
                    }
                    let configText = '当前配置:\\n';
                    if (Object.keys(displayConfig).length === 0) {
                        configText += '(暂无配置)';
                    } else {
                        for (const [key, value] of Object.entries(displayConfig)) {
                            configText += key + ': ' + (value || '(未设置)') + '\\n';
                        }
                    }
                    document.getElementById('currentConfig').textContent = configText;

                    // 更新表单值
                    document.getElementById('wkRegion').value = config.wk || '';
                    document.getElementById('ev').checked = config.ev !== 'no';
                    document.getElementById('et').checked = config.et === 'yes';
                    document.getElementById('ex').checked = config.ex === 'yes';
                    document.getElementById('ech').checked = config.ech === 'yes';
                    document.getElementById('tp').value = config.tp || '';
                    if (document.getElementById('customDNS')) document.getElementById('customDNS').value = config.customDNS || '';
                    if (document.getElementById('customECHDomain')) document.getElementById('customECHDomain').value = config.customECHDomain || '';
                    document.getElementById('scu').value = config.scu || '';
                    document.getElementById('ena').checked = config.ena === 'yes';
                    document.getElementById('epd').checked = config.epd !== 'no';
                    document.getElementById('epi').checked = config.epi !== 'no';
                    document.getElementById('egi').checked = config.egi !== 'no';
                    document.getElementById('customPath').value = config.d || '';
                    document.getElementById('customIP').value = config.p || '';
                    document.getElementById('yx').value = config.yx || '';
                    document.getElementById('yxURL').value = config.yxURL || '';
                    document.getElementById('socksConfig').value = config.s || '';

                    });

                        showStatus('KV存储未配置，无法保存配置。请先在Cloudflare Workers中配置KV存储。', 'error');
                        return;
                    }
                    if (!response.ok) throw new Error(await response.text());
                    const result = await response.json();
                    showStatus(result.message, result.success ? 'success' : 'error');
                    if (result.success) {
                        await loadCurrentConfig();
                        updateWkRegionState();
                        setTimeout(function() { window.location.reload(); }, 1500);
                    }
                } catch (error) {
                    showStatus('保存失败: ' + error.message, 'error');
                }
            }

            function showStatus(message, type) {
                const statusDiv = document.getElementById('statusMessage');
                if (statusDiv) {
                    statusDiv.textContent = message;
                    statusDiv.style.display = 'block';
                    statusDiv.style.color = type === 'success' ? '#00f0ff' : '#ff3860';
                    statusDiv.style.borderColor = type === 'success' ? '#00f0ff' : '#ff3860';
                    setTimeout(function() { statusDiv.style.display = 'none'; }, 3000);
                }
                if (typeof window.flashActionStatus === 'function') {
                    window.flashActionStatus(message, type === 'success' ? 'ok' : 'err');
                }
            }

            async function resetAllConfig() {
                if (confirm('确定要重置所有配置吗？这将清空所有KV配置，恢复为环境变量设置。')) {
                    try {
                        const response = await fetch(window.location.pathname + '/api/config', {
                            method: 'POST',
                            headers: { 'Content-Type': 'application/json' },
                            body: JSON.stringify({ 
                                wk: '', d: '', p: '', yx: '', yxURL: '', s: '', ae: '',
                                rm: '', qj: '', dkby: '', yxby: '', ev: '', et: '', ex: '', tp: '', scu: '', epd: '', epi: '', egi: '',
                                ipv4: '', ipv6: '', ispMobile: '', ispUnicom: '', ispTelecom: '', homepage: ''
                            })
                        });
                        if (response.status === 503) {
                            showStatus('KV存储未配置，无法重置配置。', 'error');
                            return;
                        }
                        if (!response.ok) throw new Error(await response.text());
                        const result = await response.json();
                        showStatus(result.message || '配置已重置', result.success ? 'success' : 'error');
                        if (result.success) {
                            await loadCurrentConfig();
                            updateWkRegionState();
                            setTimeout(function() { window.location.reload(); }, 1500);
                        }
                    } catch (error) {
                        showStatus('重置失败: ' + error.message, 'error');
                    }
                }
            }

            document.addEventListener('DOMContentLoaded', function() {
                createMatrixRain();
                checkSystemStatus();
                checkKVStatus();
                checkECHStatus();

                const echCheckbox = document.getElementById('ech');
                const portControl = document.getElementById('portControl');
                if (echCheckbox && portControl) {
                    echCheckbox.addEventListener('change', function() {
                        if (this.checked) portControl.value = 'yes';
                    });
                    if (echCheckbox.checked) portControl.value = 'yes';
                }

                const customIPInput = document.getElementById('customIP');
                if (customIPInput) customIPInput.addEventListener('input', updateWkRegionState);

                ['regionForm', 'otherConfigForm', 'advancedConfigForm'].forEach(function(fid) {
                    const f = document.getElementById(fid);
                    if (f) f.addEventListener('submit', function(e) { e.preventDefault(); });
                });

                document.querySelectorAll('#configContent input[type="text"], #configContent input[type="number"]').forEach(function(el) {
                    el.addEventListener('keydown', function(e) {
                        if (e.key === 'Enter') {
                            e.preventDefault();
                            saveAllConfig();
                        }
                    });
                });

                function collectAllConfig() {
                    function val(id) { const el = document.getElementById(id); return el ? el.value : ''; }
                    function chk(id, def) { const el = document.getElementById(id); if (!el) return def; return el.checked ? 'yes' : 'no'; }
                    return {
                        wk: val('wkRegion'),
                        ev: chk('ev', 'yes'), et: chk('et', 'no'), ex: chk('ex', 'no'), ech: chk('ech', 'no'),
                        tp: val('tp'),
                        customDNS: val('customDNS'),
                        customECHDomain: val('customECHDomain'),
                        d: val('customPath'),
                        p: val('customIP'),
                        yx: val('yx'),
                        yxURL: val('yxURL'),
                        s: val('socksConfig'),
                        homepage: val('customHomepage'),
                        scu: val('scu'),
                        ena: chk('ena', 'no'),
                        epd: chk('epd', 'yes'), epi: chk('epi', 'yes'), egi: chk('egi', 'yes')
                    };
                }

                async function saveAllConfig() {
                    const evEl = document.getElementById('ev'), etEl = document.getElementById('et'), exEl = document.getElementById('ex');
                    if (evEl && etEl && exEl && !evEl.checked && !etEl.checked && !exEl.checked) {
                        flashActionStatus('至少需要启用一个协议！', 'err');
                        showToast('至少需要启用一个协议！', 'warn');
                        return;
                    }
                    const btn = document.getElementById('cpBtnSaveAll');
                    if (btn) { btn.classList.add('cp-action-btn-saving'); btn.disabled = true; }
                    try {
                        await saveConfig(collectAllConfig());
                    } finally {
                        if (btn) { btn.classList.remove('cp-action-btn-saving'); btn.disabled = false; }
                    }
                }
                window.saveAllConfig = saveAllConfig;

                function flashActionStatus(msg, type) {
                    const el = document.getElementById('cpActionStatus');
                    if (!el) return;
                    el.textContent = msg;
                    el.classList.toggle('cp-err', type === 'err');
                    el.classList.add('cp-show');
                    clearTimeout(flashActionStatus._t);
                    flashActionStatus._t = setTimeout(function() { el.classList.remove('cp-show'); }, 2400);
                }
                window.flashActionStatus = flashActionStatus;

                const cpActionBar = document.getElementById('cpActionBar');
                const cpBtnSaveAll = document.getElementById('cpBtnSaveAll');
                if (cpBtnSaveAll) cpBtnSaveAll.addEventListener('click', async function() {
                    cpBtnSaveAll.classList.add('cp-action-btn-saving');
                    try {
                        await saveAllConfig();
                        if (cpActionBar) cpActionBar.classList.remove('cp-dirty');
                    } finally {
                        cpBtnSaveAll.classList.remove('cp-action-btn-saving');
                    }
                });
                const cpBtnRefresh = document.getElementById('cpBtnRefresh');
                if (cpBtnRefresh) cpBtnRefresh.addEventListener('click', async function() {
                    cpBtnRefresh.classList.add('cp-action-btn-saving');
                    try {
                        await loadCurrentConfig();
                        if (cpActionBar) cpActionBar.classList.remove('cp-dirty');
                        flashActionStatus('配置已刷新');
                    } finally {
                        cpBtnRefresh.classList.remove('cp-action-btn-saving');
                    }
                });
                const cpBtnReset = document.getElementById('cpBtnReset');
                if (cpBtnReset) cpBtnReset.addEventListener('click', resetAllConfig);

                function markDirty() {
                    if (cpActionBar) cpActionBar.classList.add('cp-dirty');
                }
                const dirtyScope = document.getElementById('configContent') || document;
                ['input', 'change'].forEach(function(evt) {
                    dirtyScope.addEventListener(evt, function(e) {
                        const tgt = e.target;
                        if (!tgt || !tgt.tagName) return;
                        const tag = tgt.tagName.toLowerCase();
                        if (tag === 'input' || tag === 'select' || tag === 'textarea') {
                            markDirty();
                        }
                    });
                });

                window.addEventListener('keydown', function(e) {
                    if ((e.ctrlKey || e.metaKey) && (e.key === 's' || e.key === 'S')) {
                        e.preventDefault();
                        if (cpBtnSaveAll && !cpBtnSaveAll.classList.contains('cp-action-btn-saving')) {
                            cpBtnSaveAll.click();
                        }
                    }
                });
            });
        </script>
    </body>
    </html>`;
    return new Response(pageHtml, { status: 200, headers: { 'Content-Type': 'text/html; charset=utf-8' } });
}
